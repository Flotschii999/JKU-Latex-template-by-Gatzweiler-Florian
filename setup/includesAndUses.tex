% \printLayout in config.tex:
%   \book   — twoside; sections open on a right-hand (odd) page.
%             A blank left page may be inserted; it stays empty (no header).
%   \normal — oneside; sections just start on the next page. No blank verso.
%
% Chapters use:  \sectionInit\section{...}
% \sectionInit is a no-op; the single page break lives in \section.
% \stdsection skips that break (abstracts, declarations, starred titles).
\makeatletter
% Blank pages from \cleardoublepage must not keep fancyhdr headers.
\renewcommand*{\cleardoublepage}{%
  \clearpage
  \if@twoside
    \ifodd\c@page
    \else
      \hbox{}\thispagestyle{empty}\newpage
      \if@twocolumn\hbox{}\newpage\fi
    \fi
  \fi
}

\ifthenelse{\equal{\printLayout}{\normal}}{%
  \@twosidefalse
  \@mparswitchfalse
  \newcommand{\clearforsection}{\clearpage}%
}{%
  \newcommand{\clearforsection}{\cleardoublepage}%
}
\makeatother
